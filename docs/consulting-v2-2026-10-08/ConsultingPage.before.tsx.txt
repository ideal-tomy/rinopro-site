function DocumentPreview({ index }: { index: number }) {
  if (index === 0)
    return (
      <div className={styles.document} aria-label="課題整理表の見本">
        <div className={styles.flowHeaders}>
          <span>業務の流れ</span>
          <span>主な課題</span>
          <span>関連部門</span>
        </div>
        <div className={styles.flowBody} aria-hidden="true">
          <div className={styles.flowDiagram}>
            <i />
            <b>↓</b>
            <i />
            <b>→</b>
            <i />
          </div>
          <div className={styles.documentLines}>
            {[0, 1, 2, 3].map((i) => (
              <span key={i} />
            ))}
          </div>
          <div className={styles.documentLines}>
            {[0, 1, 2, 3].map((i) => (
              <span key={i} />
            ))}
          </div>
        </div>
      </div>
    );
  if (index === 1)
    return (
      <div className={styles.document}>
        <table aria-label="優先順位表の見本">
          <thead>
            <tr>
              {["No", "取り組み内容", "効果", "実現しやすさ", "優先度"].map(
                (t) => (
                  <th key={t}>{t}</th>
                ),
              )}
            </tr>
          </thead>
          <tbody>
            {[1, 2, 3].map((n) => (
              <tr key={n}>
                <th>{n}</th>
                <td>
                  <span className={styles.line} aria-label="項目例" />
                </td>
                <td>
                  <span className={styles.meter} aria-label="比較例" />
                </td>
                <td>
                  <span className={styles.circle} aria-label="比較例" />
                </td>
                <td>
                  <span className={styles.blueCircle} aria-label="比較例" />
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    );
  return (
    <div className={styles.document}>
      <table className={styles.schedule} aria-label="実行・検証計画の見本">
        <thead>
          <tr>
            {["", "4月", "5月", "6月", "7月"].map((t, i) => (
              <th key={i}>{t}</th>
            ))}
          </tr>
        </thead>
        <tbody>
          {["試す範囲の設定", "開発・確認", "テスト・検証", "運用開始"].map(
            (t, i) => (
              <tr key={t}>
                <th>{t}</th>
                {[0, 1, 2, 3].map((j) => (
                  <td key={j}>
                    {(j === i || (i > 0 && j === i - 1)) && (
                      <span className={styles.gantt} />
                    )}
                  </td>
                ))}
              </tr>
            ),
          )}
        </tbody>
      </table>
    </div>
  );
}
