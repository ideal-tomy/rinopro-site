export function ConsultingExamples() {
  const [selected, setSelected] = useState(0);
  const tabs = useRef<(HTMLButtonElement | null)[]>([]);
  return (
    <>
      <div className={styles.tabs} role="tablist" aria-label="相談内容">
        {examples.map((example, index) => (
          <button
            key={example.label}
            type="button"
            role="tab"
            id={`example-tab-${index}`}
            aria-selected={selected === index}
            aria-controls={`example-panel-${index}`}
            tabIndex={selected === index ? 0 : -1}
            ref={(node) => {
              tabs.current[index] = node;
            }}
            onClick={() => setSelected(index)}
            onKeyDown={(event) => {
              let next: number;
              if (event.key === "ArrowRight" || event.key === "ArrowDown")
                next = (index + 1) % examples.length;
              else if (event.key === "ArrowLeft" || event.key === "ArrowUp")
                next = (index + examples.length - 1) % examples.length;
              else if (event.key === "Home") next = 0;
              else if (event.key === "End") next = examples.length - 1;
              else return;
              event.preventDefault();
              setSelected(next);
              tabs.current[next]?.focus();
            }}
          >
            {example.label}
          </button>
        ))}
      </div>
      {examples.map((example, index) => (
        <div
          key={example.label}
          id={`example-panel-${index}`}
          role="tabpanel"
          aria-labelledby={`example-tab-${index}`}
          tabIndex={0}
          hidden={selected !== index}
          className={styles.examplePanel}
        >
          <div>
            <p className={styles.situation}>相談の状況</p>
            <h3>{example.title}</h3>
            <p>{example.situation}</p>
            <p className={styles.note}>
              相談の進め方を示す例です。実績ではありません。
            </p>
          </div>
          <dl className={styles.exampleDetails}>
            {example.details.map((body, i) => (
              <div key={detailLabels[i]}>
                <dt>{detailLabels[i]}</dt>
                <dd>{body}</dd>
              </div>
            ))}
          </dl>
        </div>
      ))}
    </>
  );
}

export function ConsultingDocuments() {
  const [selected, setSelected] = useState(0);
  const dialog = useRef<HTMLDialogElement>(null);
  const closeButton = useRef<HTMLButtonElement>(null);
  const trigger = useRef<HTMLButtonElement | null>(null);
  const previousOverflow = useRef("");
  useEffect(() => {
    const element = dialog.current;
    return () => {
      if (element?.open)
        window.document.body.style.overflow = previousOverflow.current;
    };
  }, []);
  const document = documents[selected];
  function close() {
    dialog.current?.close();
  }
  return (
    <>
      <div className={styles.documentGrid}>
        {documents.map((item, i) => (
          <button
            key={item.title}
            type="button"
            aria-haspopup="dialog"
            aria-label={`${item.title}を拡大`}
            onClick={(event) => {
              trigger.current = event.currentTarget;
              setSelected(i);
              previousOverflow.current = window.document.body.style.overflow;
              window.document.body.style.overflow = "hidden";
              dialog.current?.showModal();
              closeButton.current?.focus();
            }}
          >
            <span className={styles.paper} aria-hidden="true">
              <span className={styles.paperTitle}>
                <DocumentTitle title={item.title} />
              </span>
              <span className={styles.paperGrid}>
                {Array.from({ length: 9 }, (_, j) => (
                  <i key={j} />
                ))}
              </span>
            </span>
            <span className={styles.documentName}>
              <DocumentTitle title={item.title} />
            </span>
            <span className={styles.documentAction}>見本を拡大 ↗</span>
          </button>
        ))}
      </div>
      <dialog
        ref={dialog}
        className={styles.modal}
        aria-labelledby="document-title"
        aria-describedby="document-note"
        onKeyDown={(event) => {
          if (event.key !== "Tab") return;
          const targets = event.currentTarget.querySelectorAll<HTMLElement>(
            'button, [tabindex="0"]',
          );
          const first = targets[0];
          const last = targets[targets.length - 1];
          if (event.shiftKey && window.document.activeElement === first) {
            event.preventDefault();
            last?.focus();
          } else if (
            !event.shiftKey &&
            window.document.activeElement === last
          ) {
            event.preventDefault();
            first?.focus();
          }
        }}
        onClose={() => {
          window.document.body.style.overflow = previousOverflow.current;
          trigger.current?.focus();
        }}
        onClick={(event) => {
          if (event.target !== event.currentTarget) return;
          const bounds = event.currentTarget.getBoundingClientRect();
          if (
            event.clientX < bounds.left ||
            event.clientX > bounds.right ||
            event.clientY < bounds.top ||
            event.clientY > bounds.bottom
          )
            close();
        }}
      >
        <div className={styles.modalHeader}>
          <h3 id="document-title">{document.title}</h3>
          <button type="button" ref={closeButton} onClick={close}>
            閉じる ×
          </button>
        </div>
        <p id="document-note" className={styles.note}>
          見本です。実際の納品物・顧客データではありません。
        </p>
        <div
          className={styles.tableScroll}
          role="region"
          aria-label={`${document.title}の表`}
          tabIndex={0}
        >
          <table>
            <thead>
              <tr>
                {document.columns.map((column) => (
                  <th scope="col" key={column}>
                    {column}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              <tr>
                {document.row.map((cell) => (
                  <td key={cell}>{cell}</td>
                ))}
              </tr>
            </tbody>
          </table>
        </div>
      </dialog>
    </>
  );
}

function DocumentTitle({ title }: { title: string }) {
  if (title !== "システム・情報の関係図") return title;
  return (
    <>
      <span className={styles.phrase}>システム・</span>
      <span className={styles.phrase}>情報の関係図</span>
    </>
  );
}
